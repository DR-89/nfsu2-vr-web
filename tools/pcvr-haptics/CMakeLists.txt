cmake_minimum_required(VERSION 3.20)
project(nfsu2_web_haptics LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The browser haptics helper requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_path(OPENXR_INCLUDE_DIR openxr/openxr.h REQUIRED)
add_library(nfsu2_web_haptics SHARED layer.cpp)
target_include_directories(nfsu2_web_haptics PRIVATE "${OPENXR_INCLUDE_DIR}")
add_executable(haptics-bridge bridge.cpp)
target_link_libraries(haptics-bridge PRIVATE ws2_32 bcrypt)
add_executable(haptics-test test.cpp)
target_include_directories(haptics-test PRIVATE "${OPENXR_INCLUDE_DIR}")
foreach(target nfsu2_web_haptics haptics-bridge haptics-test)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
endforeach()
enable_testing()
file(GENERATE OUTPUT "$<TARGET_FILE_DIR:nfsu2_web_haptics>/implicit-layer.json" CONTENT "{\n  \"file_format_version\": \"1.0.0\",\n  \"api_layer\": {\n    \"name\": \"XR_APILAYER_NFSU2_web_haptics\",\n    \"library_path\": \"./nfsu2_web_haptics.dll\",\n    \"api_version\": \"1.0\",\n    \"implementation_version\": \"3\",\n    \"description\": \"NFSU2 browser controller vibration\",\n    \"disable_environment\": \"NFSU2_WEB_HAPTICS_DISABLE\"\n  }\n}\n")
add_test(NAME haptics-layer COMMAND haptics-test $<TARGET_FILE:nfsu2_web_haptics>)
file(GENERATE OUTPUT "$<TARGET_FILE_DIR:nfsu2_web_haptics>/layer.json" CONTENT "{\n  \"file_format_version\": \"1.0.0\",\n  \"api_layer\": {\n    \"name\": \"XR_APILAYER_NFSU2_web_haptics\",\n    \"library_path\": \"./nfsu2_web_haptics.dll\",\n    \"api_version\": \"1.0\",\n    \"implementation_version\": \"1\",\n    \"description\": \"NFSU2 browser controller vibration\"\n  }\n}\n")
